TweenEngine — 60fps 插值引擎（~50 行）

// generic_renderer.js 内部
// 原理：引擎每秒推 2-4 个 event。TweenEngine 在 event 之间做插值。

const TweenEngine = {
  targets: {},
  lastEventTs: 0,

  onEvent(event) {
    for (const e of event.entities || []) {
      const sprite = this.getSprite(e.id);
      if (!sprite) continue;
      const duration = event.ts - this.lastEventTs > 0
        ? (event.ts - this.lastEventTs) * 1000 : 300;  // ms
      this.targets[e.id] = {
        sprite,
        fromX: sprite.x, fromY: sprite.y,
        toX:   e.pos[0] * TILE_SIZE,
        toY:   e.pos[1] * TILE_SIZE,
        startTime: performance.now(),
        duration
      };
      // 同时通知动画状态机：角色在移动
      AnimationState.setState(e.id, 'walk_right');
    }
    this.lastEventTs = event.ts;
  },

  tick(now) {
    for (const [id, t] of Object.entries(this.targets)) {
      const elapsed = now - t.startTime;
      const progress = Math.min(1, elapsed / t.duration);
      const eased = 1 - (1 - progress) ** 3;  // easeOutCubic
      t.sprite.x = t.fromX + (t.toX - t.fromX) * eased;
      t.sprite.y = t.fromY + (t.toY - t.fromY) * eased;
      if (progress >= 1) {
        delete this.targets[id];
        AnimationState.setState(id, 'idle');  // 到达目标 → 静止
      }
    }
  },

  getSprite(id) { return /* 从 EntitiesLayer.sprites 查找 */; }
};

AnimationState — 动画状态机（~40 行）

const AnimationState = {
  states: {},

  setState(entityId, newState, visualCfg) {
    if (this.states[entityId]?.currentState === newState) return;
    const animCfg = visualCfg.animations?.[entityId];
    if (!animCfg) return;  // 无动画配置 → 保持静态 sprite

    const frames = animCfg[newState];
    if (!frames) return;

    const sheet = visualCfg.sprites[entityId]?.sheet;
    const sprite = this.states[entityId]?.sprite;
    if (sprite && sheet) {
      sprite.textures = frames.map(f =>
        new PIXI.Texture(
          PIXI.BaseTexture.from(sheet),
          new PIXI.Rectangle(f[0] * TILE_SIZE, f[1] * TILE_SIZE, TILE_SIZE, TILE_SIZE)
        )
      );
      if (sprite.play) { sprite.animationSpeed = animCfg.speed; sprite.play(); }
    }
    this.states[entityId] = { sprite, currentState: newState, frames };
  },

  // 触发时机:
  //   event.action_text → setState(id, "talk")       // agent 在说话
  //   event.phase==="sensory" → setState(id, "idle")  // 只是感官更新
  //   TweenEngine 新坐标 → setState(id, "walk_right")  // 在移动
};

import { TilemapLayer }    from './layers/tilemap.js';
import { DecorationsLayer } from './layers/decorations.js';   // 新增
import { EntitiesLayer }   from './layers/entities.js';
import { DialogueLayer }   from './layers/dialogue_bubbles.js';
import { HudLayer }        from './layers/hud.js';

const layers = [
  TilemapLayer,        // Layer 0: 底板
  DecorationsLayer,    // Layer 1: 环境装饰（可删）
  EntitiesLayer,       // Layer 2: 角色/物品
  DialogueLayer,       // Layer 3: 对话气泡
  HudLayer,            // Layer 4: HUD + tooltip + freeze 按钮
];

function init(app, visualCfg, worldCfg) {
  layers.forEach(l => l.init(app, visualCfg, worldCfg));
  app.ticker.add(tick);  // 60fps render loop
}
function tick(dt) {
  const now = performance.now();
  TweenEngine.tick(now);
  layers.forEach(l => l.tick && l.tick(dt));
}
function onEvent(event) {
  TweenEngine.onEvent(event);
  layers.forEach(l => l.update && l.update(event));
}
